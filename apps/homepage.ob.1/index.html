APIs you want to read.

Domain logic lives on the model — not spread across controllers and “service layers” that only forward calls.

ottabase/models/Todo.ts 
import { BaseModel } from '@ottabase/ottaorm';
import { todosTable } from '../db/schema';

export class Todo extends BaseModel {
  static entity = 'todos';
  static table  = todosTable;

  async user() {
    const { User } = await import('@ottabase/ottaorm');
    return this.belongsTo(User, 'userId');
  }

  async toggle() {
    this.set('completed', !this.get('completed'));
    return this.save();
  }
}
import { createModelHooks } from '@ottabase/ottaorm/client';

export const {
  useList: useTodos,
  useCreate: useCreateTodo,
} = createModelHooks({ entityName: 'todos' });

const { data } = useTodos();
// GET /api/ottaorm/todos — no hand-written router.
await initRLS({
  organizationId: session.orgId,
  userId: session.user.id,
});

// Todo.all() is tenant-scoped. Full stop.
const rows = await Todo.all();
# One POST bootstraps schema — no migration dir sprawl.
